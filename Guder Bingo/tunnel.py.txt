import urllib.request
import json

print("Requesting public tunnel...")
try:
    req = urllib.request.urlopen("https://localtunnel.me/?port=3000")
    res = json.loads(req.read().decode("utf-8"))
    print("\n================================")
    print("YOUR HTTPS URL:", res.get("url"))
    print("================================\n")
except Exception as e:
    print("Error:", e)
